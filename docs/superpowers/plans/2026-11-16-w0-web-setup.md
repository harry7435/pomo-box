`tests/core-boundary.test.ts`:

```ts
// @vitest-environment node
import { ESLint } from 'eslint'
import { describe, expect, it } from 'vitest'

const BOUNDARY_RULES = new Set(['no-restricted-imports', 'no-restricted-globals'])
const eslint = new ESLint()

async function boundaryRuleIds(code: string, filePath: string): Promise<string[]> {
  const [result] = await eslint.lintText(code, { filePath })
  return result.messages
    .map((message) => message.ruleId)
    .filter((ruleId): ruleId is string => ruleId !== null && BOUNDARY_RULES.has(ruleId))
}

describe('src/core boundary', { timeout: 30_000 }, () => {
  it.each([
    ['react', "import 'react'\n"],
    ['react-dom/client', "import 'react-dom/client'\n"],
    ['dexie', "import 'dexie'\n"],
    ['firebase/firestore', "import 'firebase/firestore'\n"],
    ['@firebase/app', "import '@firebase/app'\n"],
  ])('rejects importing %s from core', async (_name, code) => {
    expect(await boundaryRuleIds(code, 'src/core/sample.ts')).toContain('no-restricted-imports')
  })

  it.each(['window', 'document', 'localStorage', 'sessionStorage', 'indexedDB', 'navigator'])(
    'rejects the browser global %s in core',
    async (name) => {
      const code = `export const value = ${name}\n`
      expect(await boundaryRuleIds(code, 'src/core/sample.ts')).toContain('no-restricted-globals')
    },
  )

  it('rejects relative imports that leave core from a top-level core file', async () => {
    const code = "import '../features/timeline'\n"
    expect(await boundaryRuleIds(code, 'src/core/sample.ts')).toContain('no-restricted-imports')
  })

  it('rejects relative imports that leave core from a core subfolder', async () => {
    const code = "import '../../features/timeline'\n"
    expect(await boundaryRuleIds(code, 'src/core/timer/sample.ts')).toContain('no-restricted-imports')
  })

  it('allows sibling imports inside core', async () => {
    const code = "import './types.ts'\n"
    expect(await boundaryRuleIds(code, 'src/core/sample.ts')).toEqual([])
  })

  it('allows a core subfolder to import from the core root', async () => {
    const code = "import '../types.ts'\n"
    expect(await boundaryRuleIds(code, 'src/core/timer/sample.ts')).toEqual([])
  })

  it('allows vitest imports in core tests', async () => {
    const code = "import { describe } from 'vitest'\nexport { describe }\n"
    expect(await boundaryRuleIds(code, 'src/core/sample.test.ts')).toEqual([])
  })

  it('does not restrict react or browser globals outside core', async () => {
    const code = "import 'react'\nexport const width = window.innerWidth\n"
    expect(await boundaryRuleIds(code, 'src/App.tsx')).toEqual([])
  })
})
```

```js
import js from '@eslint/js'
import globals from 'globals'
import reactHooks from 'eslint-plugin-react-hooks'
import reactRefresh from 'eslint-plugin-react-refresh'
import tseslint from 'typescript-eslint'
import { defineConfig, globalIgnores } from 'eslint/config'

const CORE_MESSAGE =
  'src/core는 웹과 앱이 공유하는 플랫폼 독립 영역입니다. React·브라우저·저장소 구현에 의존하지 마세요.'
const CORE_ESCAPE_MESSAGE = 'src/core 밖의 모듈을 import하지 마세요. 필요한 기능은 인터페이스로 주입받으세요.'

const corePackagePattern = {
  group: [
    'react',
    'react/*',
    'react-dom',
    'react-dom/*',
    'dexie',
    'dexie/*',
    'firebase',
    'firebase/*',
    '@firebase/*',
  ],
  message: CORE_MESSAGE,
}

const coreRestrictedGlobals = [
  'window',
  'document',
  'localStorage',
  'sessionStorage',
  'indexedDB',
  'navigator',
].map((name) => ({ name, message: CORE_MESSAGE }))

function coreBoundary(files, escapePattern) {
  return {
    files,
    rules: {
      'no-restricted-imports': [
        'error',
        { patterns: [corePackagePattern, { group: [escapePattern], message: CORE_ESCAPE_MESSAGE }] },
      ],
      'no-restricted-globals': ['error', ...coreRestrictedGlobals],
    },
  }
}

export default defineConfig([
  globalIgnores(['dist']),
  {
    files: ['**/*.{ts,tsx}'],
    extends: [
      js.configs.recommended,
      tseslint.configs.recommended,
      reactHooks.configs.flat.recommended,
      reactRefresh.configs.vite,
    ],
    languageOptions: {
      globals: globals.browser,
    },
  },
  // core 최상위 파일: core 밖으로 나가는 '../' 금지
  coreBoundary(['src/core/*.{ts,tsx}'], '../*'),
  // core 1단계 하위 폴더: core 밖으로 나가는 '../../' 금지
  coreBoundary(['src/core/*/*.{ts,tsx}'], '../../*'),
])
```

- id는 UUID, 모든 레코드에 updated_at과 deleted_at(소프트 삭제).
- 저장 스키마 변경은 버전 관리로만(웹 Dexie version(), 앱 Drizzle 마이그레이션).

```markdown
# ADR-0001: 웹 우선, React + Vite, 게스트 로컬 저장 → 로그인 시 서버 이전

- 상태: 채택
- 날짜: 2026-11-16

## 맥락

처음 계획은 React Native(Expo) 앱을 서버 없이 먼저 출시하는 것이었다. 하지만 스토어 심사와 Google 개인 계정의 비공개 테스트(12명·14일) 없이 **빨리 직접 써보며 기획을 검증하고 싶다**는 요구가 생겼다. 웹과 앱은 우선 독립적으로 동작하고, 동기화는 나중에 붙인다.

## 선택지

1. **Expo 하나로 웹·앱 동시 개발(react-native-web)**: 앱 단계 비용이 가장 작고 첫날부터 RN을 배운다. 대신 웹 출시가 느리고 웹 품질(DOM 마크업, 접근성, 번들 크기)이 떨어진다. 웹 SQLite는 alpha.
2. **React + Vite 웹, 앱은 나중에 Expo**: 익숙한 스택으로 가장 빨리 실사용. 앱 단계에서 UI를 다시 만든다. 순수 로직은 공유 core로 재사용.
3. **Next.js 웹**: 2와 같지만 SSR·서버 기능이 이 서비스(개인 데이터 SPA)에 필요 없고 기존 프로젝트와 겹친다.
4. **Next.js + Expo 공유 UI(Solito 등)**: 학습 범위는 넓지만 혼자 하는 학습 프로젝트에는 설정 부담이 과하다.

## 결정

2번. React + Vite로 웹을 먼저 출시한다.

- 공유할 로직은 `src/core/`에 모으고 ESLint로 플랫폼 의존을 막는다. A0에서 `packages/core`로 옮긴다.
- 저장은 로컬 우선: 로그인 전에는 Dexie(IndexedDB), W3에서 Firebase Auth + Firestore(오프라인 캐시)를 붙이고 첫 로그인 시 게스트 데이터를 한 번 업로드한다. 화면은 저장소 인터페이스에만 의존한다.

## 결과

- 며칠 안에 실사용 가능한 웹을 배포할 수 있다.
- 앱 단계에서 UI 재구현 비용이 생긴다. 웹에서 검증된 기능만 옮겨 비용을 줄인다.
- core 경계가 무너지면 앱 단계 재사용이 어려워지므로 lint 규칙과 테스트로 지킨다.
- 브라우저 저장소는 지워질 수 있어, W3 전까지는 데이터 유실 위험을 감수한다.
```
